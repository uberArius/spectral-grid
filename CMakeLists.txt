cmake_minimum_required(VERSION 3.24)
project(SpectralGrid VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(JSON_Install OFF CACHE BOOL "Do not package development headers" FORCE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
FetchContent_Declare(imgui URL https://github.com/ocornut/imgui/archive/refs/tags/v1.91.9b.tar.gz URL_HASH SHA256=8e1bbc76c71d74fef2fb85db7e7ca8eba13d6a86623c54992b60162db554ffdb)
FetchContent_MakeAvailable(json imgui)
add_library(spectral_core src/Config.cpp src/SpectrumAnalyzer.cpp src/GridSimulation.cpp)
target_include_directories(spectral_core PUBLIC src)
target_link_libraries(spectral_core PUBLIC nlohmann_json::nlohmann_json)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
  target_compile_options(spectral_core PRIVATE /utf-8 /W4 /permissive-)
endif()
if(WIN32)
  enable_language(RC)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_definitions(spectral_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_library(imgui_win STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp)
  target_include_directories(imgui_win PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  file(READ shaders/points.hlsl POINT_SHADER)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/points.hlsl")
  configure_file(src/ShaderSource.h.in generated/ShaderSource.h @ONLY)
  add_executable(SpectralGrid WIN32 src/main.cpp src/Application.cpp src/WindowManager.cpp src/AudioCapture.cpp src/Renderer.cpp src/Camera.cpp src/SettingsUI.cpp resources/app.rc resources/app.manifest)
  target_include_directories(SpectralGrid PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated resources)
  target_link_libraries(SpectralGrid PRIVATE spectral_core imgui_win d3d11 dxgi dcomp d3dcompiler dwmapi ole32 uuid avrt winmm shell32)
  install(TARGETS SpectralGrid RUNTIME DESTINATION .)
  install(FILES README.md TEST_REPORT.md ARCHITECTURE.md LICENSE NOTICE THIRD_PARTY_NOTICES.md CHANGELOG.md CONTRIBUTING.md DESTINATION .)
  install(DIRECTORY docs/ DESTINATION docs)
  install(FILES ${imgui_SOURCE_DIR}/LICENSE.txt DESTINATION licenses RENAME DearImGui.txt)
  install(FILES ${json_SOURCE_DIR}/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json.txt)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(spectral_tests tests/core_tests.cpp)
  target_link_libraries(spectral_tests PRIVATE spectral_core)
  if(WIN32)
    target_sources(spectral_tests PRIVATE src/Camera.cpp)
  endif()
  add_test(NAME spectral_core COMMAND spectral_tests)
endif()
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "SpectralGrid-1.0.0-Windows-x64")
include(CPack)
